// lib/assistant/openai.ts

import { assistantConfig } from './config'
import { ASSISTANT_INSTRUCTIONS } from './prompt'

export type AssistantMessage = {
  role: 'user' | 'assistant'
  content: string
}

export class AssistantProviderError extends Error {
  constructor(public readonly code: 'not-configured' | 'unavailable') {
    super(code)
  }
}

export async function generateAssistantReply(
  messages: AssistantMessage[],
): Promise<string> {
  if (!assistantConfig.openAiApiKey) {
    throw new AssistantProviderError('not-configured')
  }

  const response = await fetch('https://api.openai.com/v1/responses', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${assistantConfig.openAiApiKey}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      model: assistantConfig.openAiModel,
      instructions: ASSISTANT_INSTRUCTIONS,
      input: messages.map((message) => ({
        role: message.role,
        content: [
          {
            type: message.role === 'assistant' ? 'output_text' : 'input_text',
            text: message.content,
          },
        ],
      })),
      max_output_tokens: 300,
      store: false,
    }),
  })

  if (!response.ok) {
    throw new AssistantProviderError('unavailable')
  }

  const data = (await response.json()) as {
    output?: Array<{
      content?: Array<{
        type?: string
        text?: string
      }>
    }>
  }

  const text = data.output
    ?.flatMap((item) => item.content ?? [])
    .find((item) => item.type === 'output_text' && typeof item.text === 'string')
    ?.text?.trim()

  if (!text) {
    throw new AssistantProviderError('unavailable')
  }

  return text
}

// app/api/assistant/message/route.ts

import { NextResponse } from 'next/server'
import { z } from 'zod'
import {
  MAX_HISTORY,
  MAX_MESSAGE_LENGTH,
} from '@/lib/assistant/config'
import {
  AssistantProviderError,
  generateAssistantReply,
} from '@/lib/assistant/openai'

export const runtime = 'nodejs'

const schema = z.object({
  messages: z
    .array(
      z.object({
        role: z.enum(['user', 'assistant']),
        content: z.string().trim().min(1).max(MAX_MESSAGE_LENGTH),
      }),
    )
    .min(1)
    .max(MAX_HISTORY),
})

export async function POST(request: Request): Promise<NextResponse> {
  let body: unknown

  try {
    body = await request.json()
  } catch {
    return NextResponse.json(
      { error: 'El mensaje no tiene un formato válido.' },
      { status: 400 },
    )
  }

  const parsed = schema.safeParse(body)

  if (!parsed.success) {
    return NextResponse.json(
      { error: 'Envía un mensaje válido.' },
      { status: 400 },
    )
  }

  try {
    const text = await generateAssistantReply(parsed.data.messages)
    return NextResponse.json({ text })
  } catch (error) {
    if (error instanceof AssistantProviderError && error.code === 'not-configured') {
      return NextResponse.json(
        { error: 'El asistente no está configurado.' },
        { status: 503 },
      )
    }

    return NextResponse.json(
      { error: 'No se pudo generar la respuesta.' },
      { status: 502 },
    )
  }
}

// lib/assistant/elevenlabs.ts

import { assistantConfig, MAX_AUDIO_BYTES } from './config'
import { AssistantProviderError } from './openai'

export async function transcribeAudio(audio: File): Promise<string> {
  if (!assistantConfig.elevenLabsApiKey) {
    throw new AssistantProviderError('not-configured')
  }

  if (!audio.size || audio.size > MAX_AUDIO_BYTES) {
    throw new AssistantProviderError('unavailable')
  }

  const form = new FormData()
  form.append('file', audio, audio.name || 'audio.webm')
  form.append('model_id', assistantConfig.elevenLabsSttModel)

  const response = await fetch('https://api.elevenlabs.io/v1/speech-to-text', {
    method: 'POST',
    headers: {
      'xi-api-key': assistantConfig.elevenLabsApiKey,
    },
    body: form,
  })

  if (!response.ok) {
    throw new AssistantProviderError('unavailable')
  }

  const data = (await response.json()) as { text?: unknown }

  if (typeof data.text !== 'string' || !data.text.trim()) {
    throw new AssistantProviderError('unavailable')
  }

  return data.text.trim()
}

// app/api/assistant/transcribe/route.ts

import { NextResponse } from 'next/server'
import { MAX_AUDIO_BYTES } from '@/lib/assistant/config'
import { AssistantProviderError } from '@/lib/assistant/openai'
import { transcribeAudio } from '@/lib/assistant/elevenlabs'

export const runtime = 'nodejs'

export async function POST(request: Request): Promise<NextResponse> {
  const form = await request.formData()
  const entry = form.get('audio')

  if (!entry || typeof entry === 'string' || typeof entry.size !== 'number') {
    return NextResponse.json(
      { error: 'Adjunta un audio.' },
      { status: 400 },
    )
  }

  if (entry.size === 0) {
    return NextResponse.json(
      { error: 'El audio está vacío.' },
      { status: 400 },
    )
  }

  if (entry.size > MAX_AUDIO_BYTES) {
    return NextResponse.json(
      { error: 'El audio supera el límite permitido.' },
      { status: 413 },
    )
  }

  try {
    const text = await transcribeAudio(entry as File)
    return NextResponse.json({ text })
  } catch (error) {
    if (error instanceof AssistantProviderError && error.code === 'not-configured') {
      return NextResponse.json(
        { error: 'La transcripción no está configurada.' },
        { status: 503 },
      )
    }

    return NextResponse.json(
      { error: 'No se pudo transcribir el audio.' },
      { status: 502 },
    )
  }
}

// componentes/asistente.tsx

'use client'

import { useRef, useState } from 'react'

type Message = {
  role: 'user' | 'assistant'
  content: string
}

export default function Assistant() {
  const [messages, setMessages] = useState<Message[]>([
    {
      role: 'assistant',
      content: 'Hola. ¿Qué quieres conocer sobre EdTech?',
    },
  ])
  const [draft, setDraft] = useState('')
  const [recording, setRecording] = useState(false)
  const [loading, setLoading] = useState(false)
  const recorder = useRef<MediaRecorder | null>(null)
  const chunks = useRef<Blob[]>([])

  async function sendMessage() {
    const content = draft.trim()
    if (!content || loading || recording) return

    const userMessage = { role: 'user' as const, content }
    const history = [...messages, userMessage].slice(-10)

    setMessages((current) => [...current, userMessage])
    setDraft('')
    setLoading(true)

    try {
      const response = await fetch('/api/assistant/message', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ messages: history }),
      })

      const data = await response.json()
      if (!response.ok) throw new Error(data.error)

      setMessages((current) => [
        ...current,
        { role: 'assistant', content: data.text },
      ])
    } finally {
      setLoading(false)
    }
  }

  async function transcribe(blob: Blob) {
    const form = new FormData()
    const extension = blob.type.includes('mp4') ? 'm4a' : 'webm'
    form.append('audio', blob, `audio.${extension}`)

    const response = await fetch('/api/assistant/transcribe', {
      method: 'POST',
      body: form,
    })

    const data = await response.json()
    if (!response.ok) throw new Error(data.error)
    setDraft((current) => `${current} ${data.text}`.trim())
  }

  async function toggleRecording() {
    if (recording) {
      recorder.current?.stop()
      return
    }

    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    const type = 'audio/webm;codecs=opus'
    const mediaRecorder = MediaRecorder.isTypeSupported(type)
      ? new MediaRecorder(stream, { mimeType: type })
      : new MediaRecorder(stream)

    chunks.current = []
    recorder.current = mediaRecorder

    mediaRecorder.ondataavailable = (event) => {
      if (event.data.size > 0) chunks.current.push(event.data)
    }

    mediaRecorder.onstop = async () => {
      const blob = new Blob(chunks.current, {
        type: mediaRecorder.mimeType || 'audio/webm',
      })
      stream.getTracks().forEach((track) => track.stop())
      recorder.current = null
      setRecording(false)
      if (blob.size > 0) await transcribe(blob)
    }

    mediaRecorder.start()
    setRecording(true)
  }

  return null
}
